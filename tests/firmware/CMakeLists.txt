cmake_minimum_required(VERSION 3.16)
project(sp_native_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
set(CORE ../../firmware/components/sp_core)
add_library(sp_core ${CORE}/sp_core.c ${CORE}/sp_wire.c ${CORE}/sp_service.c ${CORE}/sp_sensors.c ${CORE}/sp_soil_analog.c)
target_include_directories(sp_core PUBLIC ${CORE}/include)
target_compile_options(sp_core PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_executable(sp_tests test_core.c)
target_link_libraries(sp_tests PRIVATE sp_core)
target_compile_options(sp_tests PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
option(SP_SANITIZE "Address and undefined behaviour sanitizers" ON)
if(SP_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(sp_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sp_core PUBLIC -fsanitize=address,undefined)
endif()
add_test(NAME domain_protocol_failures COMMAND sp_tests)
add_executable(sp_ap_policy_tests test_ap_policy.c ../../firmware/components/sp_platform/sp_ap_policy.c)
target_include_directories(sp_ap_policy_tests PRIVATE ../../firmware/components/sp_platform)
target_link_libraries(sp_ap_policy_tests PRIVATE sp_core)
target_compile_options(sp_ap_policy_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME setup_ap_policy COMMAND sp_ap_policy_tests)
set(SP_MBEDTLS_SOURCE "" CACHE PATH "Standalone Mbed TLS source directory, for native AEAD tests")
if(SP_MBEDTLS_SOURCE)
