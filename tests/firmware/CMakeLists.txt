cmake_minimum_required(VERSION 3.16)
project(sp_native_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
set(CORE ../../firmware/components/sp_core)
add_library(sp_core ${CORE}/sp_core.c ${CORE}/sp_wire.c ${CORE}/sp_service.c ${CORE}/sp_sensors.c ${CORE}/sp_soil_analog.c)
target_include_directories(sp_core PUBLIC ${CORE}/include)
target_compile_options(sp_core PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_executable(sp_tests test_core.c)
target_link_libraries(sp_tests PRIVATE sp_core)
target_compile_options(sp_tests PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
option(SP_SANITIZE "Address and undefined behaviour sanitizers" ON)
if(SP_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(sp_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sp_core PUBLIC -fsanitize=address,undefined)
endif()
add_test(NAME domain_protocol_failures COMMAND sp_tests)
set(SP_MBEDTLS_SOURCE "" CACHE PATH "Standalone Mbed TLS source directory, for native AEAD tests")
if(SP_MBEDTLS_SOURCE)
  get_filename_component(SP_IDF_COMPONENTS "${SP_MBEDTLS_SOURCE}/../.." ABSOLUTE)
  add_library(sp_json ${SP_IDF_COMPONENTS}/json/cJSON/cJSON.c)
  target_include_directories(sp_json PUBLIC ${SP_IDF_COMPONENTS}/json/cJSON)
  add_executable(sp_api_tests test_api.c ../../firmware/components/sp_api/sp_api.c)
  target_include_directories(sp_api_tests PRIVATE ../../firmware/components/sp_api/include)
  target_link_libraries(sp_api_tests PRIVATE sp_core sp_json m)
  target_compile_options(sp_api_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME api_sensor_contract COMMAND sp_api_tests)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(DISABLE_PACKAGE_CONFIG_AND_INSTALL ON CACHE BOOL "" FORCE)
  add_subdirectory(${SP_MBEDTLS_SOURCE} mbedtls EXCLUDE_FROM_ALL)
  add_executable(sp_crypto_tests test_crypto.c ../../firmware/components/sp_platform/sp_crypto.c)
  target_include_directories(sp_crypto_tests PRIVATE ../../firmware/components/sp_platform/include)
  target_link_libraries(sp_crypto_tests PRIVATE sp_core mbedcrypto)
  target_compile_options(sp_crypto_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME frame_authentication_replay COMMAND sp_crypto_tests)
endif()
